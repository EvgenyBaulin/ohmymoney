archive-1.0.0.zip and archive-1.0.0-sealed.zip: transfer archives written by the code of the
v1.0.0 tag, for OlderArchiveFixtureTests. Synthetic data only: the 1.0.0 sample generator
(seed 1000, one month ending 2026-09-20, Russian names) plus a few rows written in SQL.

How they were made, from the repository root:

  git archive v1.0.0 Packages/AppCore Packages/AppDatabase Schema \
    Apps/macOS/Platform/Archive/CryptoKitArchiveCipher.swift | tar -x -C <work>/v100
  # <work>/v100maker/Package.swift and Sources/maker/main.swift as below, plus a copy of
  # CryptoKitArchiveCipher.swift in Sources/maker
  swift run --package-path <work>/v100maker maker <work>/v100/Schema <output>

The sealed one's password is «пароль 1.0.0», with 1 000 iterations and a fixed salt and nonce.

---- Sources/maker/main.swift ----
import AppCore
import AppDatabase
import CoreKit
import Foundation
import GRDB

// Writes two transfer archives with the code of the v1.0.0 tag, the way 1.0.0's ArchiveService
// wrote them: a snapshot through ExportRepository.snapshot, its CSV files, the settings JSON,
// ArchiveBuilder — plain, and sealed with a password through CryptoKit's AES-GCM.
// Usage: maker <schema dir> <output dir>
let arguments = CommandLine.arguments
let schema = DirectorySchemaSource(directory: URL(fileURLWithPath: arguments[1], isDirectory: true))
let output = URL(fileURLWithPath: arguments[2], isDirectory: true)
let work = FileManager.default.temporaryDirectory.appendingPathComponent("v100maker-\(UUID().uuidString)")
try FileManager.default.createDirectory(at: work, withIntermediateDirectories: true)
defer { try? FileManager.default.removeItem(at: work) }
let url = work.appendingPathComponent("finance.sqlite")

// Small pages keep the fixture small: set before the first table is made.
do {
  let queue = try DatabaseQueue(path: url.path)
  try queue.writeWithoutTransaction { db in try db.execute(sql: "PRAGMA page_size = 1024; VACUUM") }
  try queue.close()
}

let stack = try DatabaseStack(url: url, schema: schema)
let calendar = CalendarContext.moscow
let set = SampleDataGenerator(seed: 1_000).generate(
  months: 1, endingOn: DateOnly(year: 2026, month: 9, day: 20), calendar: calendar, language: "ru",
  density: 1)
try TransactionRepository(writer: stack.writer).insert(HistoryBatch(sample: set))

// What a 1.0.0 history held beyond the sample: an account with a lower-case currency and none
// flagged main, an operation without an account, the planning book, rates, a template, a
// choice of the model and an anomaly waved away, settings.
try stack.writer.write { db in
  try db.execute(sql: "UPDATE payment_methods SET is_default = 0")
  try db.execute(
    sql: "UPDATE payment_methods SET currency = 'usd' WHERE rowid = (SELECT MIN(rowid) FROM payment_methods)")

    db, sql: "SELECT id FROM categories WHERE kind = 'expense' AND parent_id IS NOT NULL LIMIT 1")!
  let income: String = try String.fetchOne(

  let operation: String = try String.fetchOne(db, sql: "SELECT id FROM transactions LIMIT 1")!
  let payment = "5C9E1A40-0000-4000-8000-000000000001"
  let expected = "5C9E1A40-0000-4000-8000-000000000002"
  try db.execute(

        reimbursable, freq, interval, day, next_date, active)

let snapshotURL = work.appendingPathComponent("snapshot.sqlite")
let snapshot = try ExportRepository(writer: stack.writer).snapshot(to: snapshotURL)
try stack.close()
let data = try Data(contentsOf: snapshotURL)
var builder = ArchiveBuilder(
  metadata: ArchiveBuilder.Metadata(
    appVersion: "1.0.0", schemaVersion: stack.applied.onDisk,
    createdAt: DateOnly(year: 2026, month: 9, day: 25), platform: "macOS",
    rowCounts: snapshot.rowCounts))
try builder.add(path: ArchivePaths.database, data: data)
for table in snapshot.tables {
  try builder.add(path: ArchivePaths.csv(table: table.name), data: table.data)
}
try builder.add(
  path: ArchivePaths.settings,
  data: try JSONSerialization.data(
    withJSONObject: ["language": "ru", "theme.scheme": "dark", "theme.accent": "green"],
    options: [.sortedKeys]))
try FileManager.default.createDirectory(at: output, withIntermediateDirectories: true)
try builder.build().write(to: output.appendingPathComponent("archive-1.0.0.zip"))

let salt = (0..<16).map { UInt8(truncatingIfNeeded: $0 &* 37 &+ 11) }
let nonce = (0..<12).map { UInt8(truncatingIfNeeded: $0 &* 53 &+ 7) }
try builder.build(
  password: "пароль 1.0.0", cipher: CryptoKitArchiveCipher(), salt: salt, nonce: nonce,
  iterations: 1_000
).write(to: output.appendingPathComponent("archive-1.0.0-sealed.zip"))
print("schema \(stack.applied.onDisk), tables \(snapshot.tables.count), bytes \(data.count)")
print(snapshot.rowCounts.sorted { $0.key < $1.key })
