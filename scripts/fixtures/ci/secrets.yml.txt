# Reads two secrets of the owner's and GitHub's own token; a third is named only in a comment.
jobs:
  release:
    runs-on: macos-26
    timeout-minutes: 120
    steps:
      - name: Build
        env:
          KEY: ${{ secrets.RELEASE_KEY }}   # not ${{ secrets.COMMENTED_KEY }}
        run: make build
      # SIGNING_KEY: ${{ secrets.COMMENTED_KEY }}
      - name: Sign
        env:
          SIGNING: ${{ secrets.SIGNING_KEY }}
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: make sign
